# Paparazzi-Worker — dependências dos dois sistemas
# Instale com:  pip install -r requirements.txt
#
# OBS: o FFmpeg NÃO é um pacote pip — instale pelo sistema:
#   sudo apt install ffmpeg
#
# Máquina SEM GPU: usamos o torch versão CPU (evita ~2GB de libs CUDA/nvidia).
# A linha abaixo aponta o índice de wheels CPU do PyTorch.
--extra-index-url https://download.pytorch.org/whl/cpu

# --- Comum / YouTube ---
google-api-python-client   # cliente da YouTube Data API (googleapiclient)
isodate                    # parse de durações/datas ISO 8601
python-dotenv              # carrega o .env
requests                   # chamadas HTTP (LLM/Ollama, TikTok API, diagnósticos)

# --- Coleta / Download ---
yt-dlp                     # download de vídeos e shorts

# --- IA / Transcrição (roteirizador + coletor do mirror) ---
faster-whisper             # transcrição do mirror (CTranslate2: +rápido/preciso em CPU)
openai-whisper             # usado pelo roteirizador e pela versão CLI do coletor
torch==2.13.0+cpu          # backend do Whisper — versão CPU (sem CUDA)
numpy                      # análise de frames (detecção da faixa da legenda)
Pillow                     # leitura de frames para detectar a legenda original
easyocr                    # OCR: localiza a legenda queimada (posição exata, qualquer estilo)
# obs: easyocr requer torchvision compatível com o torch instalado (CPU):
#   pip install torchvision --index-url https://download.pytorch.org/whl/cpu

# --- Mirror-Clips: postagem no TikTok (Selenium) ---
selenium                   # automação de navegador
webdriver-manager          # baixa/gerencia o chromedriver

# --- Detecção no Twitter/X (experimental) ---
tweepy                     # API do Twitter/X

# --- Painel de Controle (Django) + Postgres ---
Django==4.2.*              # painel local (dashboard + admin) e ORM único
dj-database-url            # lê DATABASE_URL do .env
psycopg2-binary            # driver Postgres
